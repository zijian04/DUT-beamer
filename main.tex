\newcommand{\ModernPlaceholder}[2][3.7cm]{%
  \begin{tikzpicture}
    \node[
      draw=DLUTLine,
      fill=DLUTMist,
      rounded corners=2pt,
      minimum width=.96\linewidth,
      minimum height=#1,
      text=DLUTMuted,
      align=center
    ] {\small #2};
  \end{tikzpicture}%
}
\newcommand{\ModernTag}[1]{%
  \tikz[baseline=(tag.base)]\node[
    fill=DLUTSky,
    text=DLUTDeep,
    rounded corners=2pt,
    inner xsep=5pt,
    inner ysep=2.1pt
  ] (tag) {\scriptsize\bfseries #1};%
}
\newcommand{\ModernMetric}[2]{%
  \begin{tikzpicture}
    \node[
      draw=DLUTLine,
      fill=DLUTMist,
      rounded corners=2pt,
      minimum width=.255\textwidth,
      minimum height=1.18cm,
      align=left,
      inner sep=7pt
    ] {%
      {\scriptsize\color{DLUTMuted}#1}\\[-1pt]%
      {\Large\bfseries\color{DLUTNavy}#2}%
    };
  \end{tikzpicture}%
}

\section{模板概览}

\begin{frame}{设计目标}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.60\textwidth}
      \begin{itemize}
        \item 面向\alert{毕业设计答辩}场景，提供简洁克制的视觉风格。
        \item 顶部无装饰条，frame title 为纯蓝色文字，最大化留白。
        \item 底部四段渐变页脚，自左向右颜色逐渐加深。
        \item 蓝色基调，配色统一，避免饱和度过高造成的廉价感。

      \vspace{.25cm}
      \begin{alertblock}{一句话总结}
        让幻灯片把注意力交还给\textbf{内容}，主题只做安静的底色。
      \end{alertblock}

\section{主题元素}

\begin{frame}{frame title 与正文样式}
  \framesubtitle{标题旁可附加副标题，使用较浅灰色}
  \begin{itemize}
    \item 一级项目以\alert{蓝色实心圆点}标记，可使用 \texttt{\textbackslash alert\{\}} 高亮。
    \begin{itemize}
      \item 二级项目自动缩进，圆点更小、颜色更深。
      \begin{itemize}
        \item 三级项目以蓝色\textbf{描边}圆环呈现。
      \end{itemize}
    \end{itemize}
    \item 数学公式继承 Beamer 默认设置：$f(x)=\sum_{i=1}^{N}\alpha_i\,\phi_i(x)$。
    \item 行内代码 / 命令使用 \texttt{\textbackslash texttt}：\texttt{python main.py}。
  \end{itemize}
\end{frame}

    \node[cell,fill=DLUTBlue!75,text=white,minimum width=1.5cm] (d) at (8.15,0) {页码};

  \vspace{.55cm}
  \begin{itemize}
    \item 四段单元宽度比为 \texttt{30 : 42 : 18 : 10}，自动占满页面宽度。
    \item 内容来源：\texttt{\textbackslash insertshortauthor / \textbackslash insertshortinstitute / \textbackslash insertshorttitle / \textbackslash insertshortdate / \textbackslash insertframenumber}。
    \item 渐变颜色由 \texttt{DLUTBlue} 不同明度生成，统一色相便于打印阅读。
  \end{itemize}
\end{frame}

\begin{frame}{占位框 / 标签 / 指标卡}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{.48\textwidth}
      本模板示例文件提供三个工具宏（在 \texttt{main.tex} 中定义）：
      \begin{itemize}
        \item \texttt{\textbackslash ModernPlaceholder[高度]\{说明\}} —— 图片占位框；
        \item \texttt{\textbackslash ModernTag\{关键词\}} —— 行内标签；
        \item \texttt{\textbackslash ModernMetric\{标题\}\{数值\}} —— 指标卡。
      \end{itemize}
      它们都使用主题预定义的蓝色色名，确保风格一致。
    \end{column}
